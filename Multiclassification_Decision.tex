\section{Problem}
% ─────────────────────────────────────────────────────────────────────────────

\subsection{Setup}

We observe a feature vector $x \in \mathcal{X} \subseteq \R^{d}$ and wish to
assign it one of $K \geq 2$ discrete labels from
$\mathcal{Y} = \{1, 2, \ldots, K\}$.
A probabilistic model supplies the class-posterior distribution
$p(y = k \mid x)$ for each class $k$.

\begin{definition}[Multiclass Decision Problem]
    Let $\mathcal{X} \subseteq \R^{d}$ be the feature space and
    $\mathcal{Y} = \{1, 2, \ldots, K\}$ the label space with $K \geq 2$ classes.
    A \emph{decision rule} is any measurable function $h : \mathcal{X} \to \mathcal{Y}$.
    The \emph{expected 0-1 loss} (Bayes risk) of $h$ is
    \[
        R(h) \;=\; \E_{(x,y)}\!\bigl[\mathbf{1}\{h(x) \neq y\}\bigr],
    \]
    where the expectation is taken over the joint distribution $p(x, y)$.
    The goal is to find the rule $h^{*}$ that minimises $R(h)$.
\end{definition}

\subsection{Assumptions}

\begin{enumerate}[label=\textbf{A\arabic*.}]
    \item \textbf{Known posteriors.} The true class-posterior probabilities
          $p(y = k \mid x)$ are available for all $k \in \mathcal{Y}$ and every $x$.
    \item \textbf{0-1 loss.} The loss incurred for predicting label $\hat{y}$ when the
          true label is $y$ is
          $\mathcal{L}(\hat{y},\, y) = \mathbf{1}\{\hat{y} \neq y\}$.
    \item \textbf{Proper probability model.} For every $x$,
          $\sum_{k=1}^{K} p(y = k \mid x) = 1$ and $p(y = k \mid x) \geq 0$.
\end{enumerate}

\subsection{Question}

\emph{Given access to the class posteriors $p(y = k \mid x)$, what decision rule
$h^{*} : \mathcal{X} \to \mathcal{Y}$ minimises the expected 0-1 loss $R(h)$?}

\subsection{Objective}

Find $h^{*}$ such that
\[
    h^{*} \;=\; \arg\min_{h\,:\,\mathcal{X}\to\mathcal{Y}} R(h)
             \;=\; \arg\min_{h} \;\E_{(x,y)}\!\bigl[\mathbf{1}\{h(x) \neq y\}\bigr].
\]

% ─────────────────────────────────────────────────────────────────────────────
\section{Derivation}
% ─────────────────────────────────────────────────────────────────────────────

The following derivation establishes the optimal multiclass decision rule
step by step.

% ── Step 1 ────────────────────────────────────────────────────────────────────
\subsection{Step 1: Decompose the Bayes Risk via the Law of Total Expectation}

The 0-1 risk $R(h)$ can be decomposed by first conditioning on $x$ and then
averaging over the marginal $p(x)$:

\begin{proof}
    \begin{align}
        R(h)
        &= \E_{(x,y)}\!\bigl[\mathbf{1}\{h(x) \neq y\}\bigr] \notag \\
        &= \E_{x}\!\Bigl[\E_{y \mid x}\!\bigl[\mathbf{1}\{h(x) \neq y\}\bigr]\Bigr].
        \label{eq:step1}
    \end{align}
    Because the outer expectation $\E_{x}[\cdot]$ is non-negative and independent
    of the choice of $h$ for any fixed $x$, minimising $R(h)$ is equivalent to
    minimising the inner conditional expectation \emph{pointwise} for each $x$.
\end{proof}

% ── Step 2 ────────────────────────────────────────────────────────────────────
\subsection{Step 2: Expand the Conditional Expected Loss for a Fixed $x$}

For a fixed $x$, the decision rule assigns a single label $h(x) = j$ for some
$j \in \mathcal{Y}$.  The conditional expected 0-1 loss at that point is:

\begin{proof}
    \begin{align}
        \E_{y \mid x}\!\bigl[\mathbf{1}\{h(x) \neq y\}\bigr]
        &= \sum_{k=1}^{K} \mathbf{1}\{j \neq k\}\, p(y = k \mid x) \notag \\
        &= 1 - p(y = j \mid x).
        \label{eq:step2}
    \end{align}
    The second equality uses $\sum_{k} p(y=k\mid x) = 1$ (Assumption \textbf{A3}),
    so $\sum_{k \neq j} p(y=k\mid x) = 1 - p(y=j\mid x)$.
\end{proof}

% ── Step 3 ────────────────────────────────────────────────────────────────────
\subsection{Step 3: Minimise Pointwise over $x$}

Since $1 - p(y = j \mid x)$ is minimised by maximising $p(y = j \mid x)$ over
$j$, the optimal label for each $x$ is:

\begin{proof}
    \[
        h^{*}(x)
        \;=\; \arg\min_{j \in \mathcal{Y}}\;\bigl[1 - p(y = j \mid x)\bigr]
        \;=\; \arg\max_{k \in \mathcal{Y}}\; p(y = k \mid x).
    \]
    Applying this choice for every $x$ and substituting back into
    \eqref{eq:step1} gives the minimal Bayes risk:
    \[
        R(h^{*})
        \;=\; \E_{x}\!\Bigl[1 - \max_{k \in \mathcal{Y}} p(y = k \mid x)\Bigr],
    \]
    known as the \emph{Bayes error rate}.
\end{proof}

% ── Final Result ──────────────────────────────────────────────────────────────
\subsection{Final Result}

\begin{theorem}[Bayes-Optimal Multiclass Decision Rule]
    Under assumptions \textbf{A1}--\textbf{A3} with a 0-1 loss, the unique
    Bayes-optimal decision rule is
    \[
        \boxed{h^{*}(x) \;=\; \arg\max_{k \in \{1,\ldots,K\}} p(y = k \mid x).}
    \]
    Its Bayes risk equals
    \(
        R(h^{*}) = \E_{x}\!\bigl[1 - \max_{k} p(y=k\mid x)\bigr].
    \)
\end{theorem}

\begin{proof}
    Follows directly from Steps 1--3.
    \begin{enumerate}
        \item (Step~1) $R(h) = \E_{x}[\E_{y|x}[\mathbf{1}\{h(x)\neq y\}]]$,
              so it suffices to minimise pointwise.
        \item (Step~2) The pointwise conditional loss equals $1 - p(y = h(x)\mid x)$.
        \item (Step~3) This is minimised by picking the class with the highest
              posterior probability, yielding $h^{*}(x) = \arg\max_k p(y=k\mid x)$.\qed
    \end{enumerate}
\end{proof}
